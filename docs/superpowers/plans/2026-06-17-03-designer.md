# Box-64 — Plan 3: Box Designer

> **For agentic workers:** REQUIRED SUB-SKILL: superpowers:executing-plans

**Goal:** Build the full 4-step box designer at `/designer` — image upload, canvas rendering of all 3 visible faces, isometric CSS 3D preview, and client-side PDF export of the flat print template.

**Architecture:** Zustand store holds `DesignState`. Each box face is a `<canvas>` element that re-renders on any state change. The isometric preview is pure CSS (3 divs + skewX/Y). PDF export renders all 6 faces onto a single A4 canvas and uses jsPDF to produce a downloadable file. No server calls in this plan.

**Tech Stack:** Zustand 5, HTML Canvas API, jsPDF, Framer Motion (step transitions), React Hook Form (Step 2 form), Tailwind CSS

```typescript
import { create } from 'zustand'
import type { DesignState, BoxSize, LogoVariant } from '@/lib/types'

interface DesignerStore extends DesignState {
  step: number
  setField: <K extends keyof DesignState>(key: K, value: DesignState[K]) => void
  setCarImageUrl: (url: string | null) => void
  nextStep: () => void
  prevStep: () => void
  reset: () => void
}

const DEFAULT_STATE: DesignState = {
  car_image_url:  null,
  car_name:       '',
  specs_line:     '1:64 Scale · Diecast',
  warning_text:   'WARNING: CHOKING HAZARD — Not for children under 3 years.',
  logo_variant:   'minigt',
  custom_logo_url: null,
  bg_color:       '#111212',
  accent_color:   '#C9A84C',
  text_color:     '#FFFFFF',
  logo_tint:      '#FFFFFF',
  box_size:       'minigt',
  quantity:       1,
}

export const useDesignerStore = create<DesignerStore>((set) => ({
  ...DEFAULT_STATE,
  step: 1,

  setField: (key, value) => set({ [key]: value }),
  setCarImageUrl: (url) => set({ car_image_url: url }),
  nextStep: () => set(s => ({ step: Math.min(4, s.step + 1) })),
  prevStep: () => set(s => ({ step: Math.max(1, s.step - 1) })),
  reset: () => set({ ...DEFAULT_STATE, step: 1 }),
}))
```

```typescript
// src/lib/canvas/boxRenderer.ts
import type { DesignState, BoxSize } from '@/lib/types'

// Pixel dimensions per box size (2× for retina sharpness)
const BOX_DIMS: Record<BoxSize, { front: [number, number]; top: [number, number]; side: [number, number] }> = {
  minigt:  { front: [480, 220], top: [480, 88],  side: [200, 220] },
  poprace: { front: [460, 240], top: [460, 96],  side: [220, 240] },
}

export function buildFrontFaceLayout(size: BoxSize) {
  const [width, height] = BOX_DIMS[size].front
  return { width, height }
}
export function buildTopFaceLayout(size: BoxSize) {
  const [width, height] = BOX_DIMS[size].top
  return { width, height }
}
export function buildSideFaceLayout(size: BoxSize) {
  const [width, height] = BOX_DIMS[size].side
  return { width, height }
}

// ─── FRONT FACE ───
export async function drawFrontFace(
  ctx: CanvasRenderingContext2D,
  state: DesignState,
  carImage: HTMLImageElement | null,
) {
  const { width, height } = buildFrontFaceLayout(state.box_size)
  ctx.clearRect(0, 0, width, height)

  // Background
  ctx.fillStyle = state.bg_color
  ctx.fillRect(0, 0, width, height)

  // Accent stripe (vertical, near right edge)
  const stripeX = width - 72
  ctx.fillStyle = state.accent_color
  ctx.globalAlpha = 0.9
  ctx.fillRect(stripeX, 8, 2, height - 16)
  ctx.fillRect(stripeX + 4, 8, 0.5, height - 16)
  ctx.globalAlpha = 1

  // Car image (left area, vertically centered)
  if (carImage) {
    const imgArea = { x: 16, y: 20, w: width * 0.55, h: height - 40 }
    const scale = Math.min(imgArea.w / carImage.width, imgArea.h / carImage.height)
    const dw = carImage.width * scale
    const dh = carImage.height * scale
    const dx = imgArea.x + (imgArea.w - dw) / 2
    const dy = imgArea.y + (imgArea.h - dh) / 2
    ctx.drawImage(carImage, dx, dy, dw, dh)
  }

  // Car name
  if (state.car_name) {
    ctx.fillStyle = state.text_color
    ctx.font = `bold ${height * 0.14}px "Barlow Condensed", sans-serif`
    ctx.textAlign = 'left'
    ctx.textBaseline = 'bottom'
    ctx.fillText(state.car_name.toUpperCase(), 14, height - 24, width * 0.75)
  }

  // Specs line
  if (state.specs_line) {
    ctx.fillStyle = state.text_color
    ctx.globalAlpha = 0.55
    ctx.font = `600 ${height * 0.08}px "Plus Jakarta Sans", sans-serif`
    ctx.textAlign = 'left'
    ctx.textBaseline = 'bottom'
    ctx.fillText(state.specs_line, 14, height - 8, width * 0.7)
    ctx.globalAlpha = 1
  }

  // Logo area (top-right, inside stripe)
  drawLogoText(ctx, state, stripeX + 10, 10, width - stripeX - 12, 60)

  // Warning (bottom-right, tiny)
  if (state.warning_text) {
    ctx.fillStyle = state.text_color
    ctx.globalAlpha = 0.2
    ctx.font = `500 ${height * 0.055}px "Plus Jakarta Sans", sans-serif`
    ctx.textAlign = 'right'
    ctx.textBaseline = 'bottom'
    ctx.fillText('⚠ ' + state.warning_text.substring(0, 60), width - 8, height - 6, 200)
    ctx.globalAlpha = 1
  }
}

// ─── TOP FACE ───
export function drawTopFace(
  ctx: CanvasRenderingContext2D,
  state: DesignState,
) {
  const { width, height } = buildTopFaceLayout(state.box_size)
  ctx.clearRect(0, 0, width, height)

  // Slightly lighter than front
  ctx.fillStyle = lighten(state.bg_color, 0.15)
  ctx.fillRect(0, 0, width, height)

  // Car name abbreviated
  if (state.car_name) {
    ctx.fillStyle = state.text_color
    ctx.globalAlpha = 0.7
    ctx.font = `bold ${height * 0.35}px "Barlow Condensed", sans-serif`
    ctx.textAlign = 'center'
    ctx.textBaseline = 'middle'
    ctx.fillText(state.car_name.toUpperCase(), width / 2, height / 2, width - 24)
    ctx.globalAlpha = 1
  }
}

// ─── SIDE FACE ───
export function drawSideFace(
  ctx: CanvasRenderingContext2D,
  state: DesignState,
) {
  const { width, height } = buildSideFaceLayout(state.box_size)
  ctx.clearRect(0, 0, width, height)

  // Darker than front
  ctx.fillStyle = darken(state.bg_color, 0.25)
  ctx.fillRect(0, 0, width, height)

  // Car name rotated vertical
  if (state.car_name) {
    ctx.save()
    ctx.fillStyle = state.text_color
    ctx.globalAlpha = 0.6
    ctx.font = `bold ${width * 0.2}px "Barlow Condensed", sans-serif`
    ctx.textAlign = 'center'
    ctx.textBaseline = 'middle'
    ctx.translate(width / 2, height / 2)
    ctx.rotate(-Math.PI / 2)
    ctx.fillText(state.car_name.toUpperCase(), 0, 0, height - 20)
    ctx.restore()
    ctx.globalAlpha = 1
  }

  // Accent bar at top edge
  ctx.fillStyle = state.accent_color
  ctx.globalAlpha = 0.6
  ctx.fillRect(0, 0, width, 2)
  ctx.globalAlpha = 1
}

// ─── LOGO TEXT HELPER ───
// Renders brand name as canvas text (not DOM) — brand protection
function drawLogoText(
  ctx: CanvasRenderingContext2D,
  state: DesignState,
  x: number, y: number, maxW: number, maxH: number
) {
  const labels: Record<string, string[]> = {
    minigt:  ['MINI', 'GT™'],
    poprace: ['POP', 'RACE'],
    custom:  [state.car_name?.split(' ')[0]?.toUpperCase() ?? ''],
  }
  const lines = labels[state.logo_variant] ?? []
  ctx.fillStyle = state.logo_tint
  ctx.textAlign = 'right'
  lines.forEach((line, i) => {
    ctx.font = `900 ${maxH * 0.38}px "Barlow Condensed", sans-serif`
    ctx.textBaseline = 'top'
    ctx.fillText(line, x + maxW, y + i * maxH * 0.4, maxW)
  })
}

// ─── COLOR HELPERS ───
function hexToRgb(hex: string): [number, number, number] {
  const n = parseInt(hex.replace('#', ''), 16)
  return [(n >> 16) & 255, (n >> 8) & 255, n & 255]
}
function rgbToHex(r: number, g: number, b: number): string {
  return '#' + [r, g, b].map(v => Math.max(0, Math.min(255, v)).toString(16).padStart(2, '0')).join('')
}
function lighten(hex: string, amt: number): string {
  const [r, g, b] = hexToRgb(hex)
  return rgbToHex(r + Math.round(amt * 255), g + Math.round(amt * 255), b + Math.round(amt * 255))
}
function darken(hex: string, amt: number): string {
  const [r, g, b] = hexToRgb(hex)
  return rgbToHex(r - Math.round(amt * 255), g - Math.round(amt * 255), b - Math.round(amt * 255))
}
```

```typescript
// src/lib/pdf/exportPdf.ts
// Client-side PDF export. Renders all 6 box faces onto an A4 canvas
// in the flat template layout (matching demo-area.svg proportions),
// then exports via jsPDF.

import type { BoxSize, DesignState } from '@/lib/types'

// SVG flat layout proportions (from demo-area.svg, scaled to 2× pixel density)
// ViewBox 841.89 × 595.28 → scale factor: canvas / viewBox
const SCALE = 2

// Face pixel rects in the flat layout — derived from SVG coords × SCALE
const FACE_RECTS = {
  minigt: {
    // SVG viewBox: 841.89 × 595.28. Scale = 2.
    // Main face (orange): x=277.88 y=314.65 w=286.22 h=130.12
    front:  { x: 556,  y: 629,  w: 572, h: 260 },
    top:    { x: 556,  y: 369,  w: 572, h: 260 },  // green (left side = top in 3D)
    bottom: { x: 556,  y: 108,  w: 572, h: 260 },  // blue-top (đáy)
    right:  { x: 556,  y: 890,  w: 572, h: 260 },  // blue-bottom (right side)
    lid_l:  { x: 296,  y: 629,  w: 260, h: 260 },  // pink left
    lid_r:  { x: 1128, y: 629,  w: 260, h: 260 },  // pink right
    canvasW: 1684, canvasH: 1191,
  },
}

export interface FaceSpec {
  name: string
  x: number
  y: number
  w: number
  h: number
}

export interface FlatLayoutSpec {
  faces: FaceSpec[]
  canvasWidth: number
  canvasHeight: number
}

export function buildFlatLayoutSpec(size: BoxSize): FlatLayoutSpec {
  const rects = FACE_RECTS[size] ?? FACE_RECTS.minigt
  return {
    faces: [
      { name: 'front',  ...pick(rects.front)  },
      { name: 'top',    ...pick(rects.top)    },
      { name: 'bottom', ...pick(rects.bottom) },
      { name: 'right',  ...pick(rects.right)  },
      { name: 'lid_l',  ...pick(rects.lid_l)  },
      { name: 'lid_r',  ...pick(rects.lid_r)  },
    ],
    canvasWidth:  rects.canvasW,
    canvasHeight: rects.canvasH,
  }
}

function pick(r: { x: number; y: number; w: number; h: number }) {
  return { x: r.x, y: r.y, w: r.w, h: r.h }
}

// ─── MAIN EXPORT ───
// Called from Step4Preview after all canvases are rendered.
// frontCanvas, topCanvas, sideCanvas — the live preview canvases.
export async function exportBoxPdf(
  state: DesignState,
  frontCanvas: HTMLCanvasElement,
  topCanvas: HTMLCanvasElement,
  sideCanvas: HTMLCanvasElement,
): Promise<void> {
  const { jsPDF } = await import('jspdf')
  const spec = buildFlatLayoutSpec(state.box_size)

  // 1. Compose all faces onto a single flat canvas
  const flatCanvas = document.createElement('canvas')
  flatCanvas.width  = spec.canvasWidth
  flatCanvas.height = spec.canvasHeight
  const ctx = flatCanvas.getContext('2d')!

  // Background white
  ctx.fillStyle = '#FFFFFF'
  ctx.fillRect(0, 0, flatCanvas.width, flatCanvas.height)

  // Draw cut guide outline (light grey dashes)
  ctx.strokeStyle = '#CCCCCC'
  ctx.setLineDash([8, 4])
  ctx.lineWidth = 1
  spec.faces.forEach(f => ctx.strokeRect(f.x, f.y, f.w, f.h))
  ctx.setLineDash([])

  // Stamp each face canvas
  const faceCanvases: Record<string, HTMLCanvasElement> = {
    front: frontCanvas, top: topCanvas, right: sideCanvas,
    // Remaining faces (bottom, lid_l, lid_r) reuse the respective face with
    // slightly adjusted tones. For this client-side preview version we tile them.
    bottom: topCanvas, lid_l: sideCanvas, lid_r: sideCanvas,
  }
  spec.faces.forEach(f => {
    const src = faceCanvases[f.name]
    if (src) ctx.drawImage(src, f.x, f.y, f.w, f.h)
  })

  // Add crop mark corners
  drawCropMarks(ctx, spec.faces)

  // 2. Export as PDF (A4 landscape at 150dpi)
  const pdf = new jsPDF({ orientation: 'landscape', unit: 'mm', format: 'a4' })
  const pdfW = pdf.internal.pageSize.getWidth()
  const pdfH = pdf.internal.pageSize.getHeight()
  const imgData = flatCanvas.toDataURL('image/png')
  pdf.addImage(imgData, 'PNG', 3, 3, pdfW - 6, pdfH - 6)
  pdf.save(`box64-${state.car_name.replace(/\s+/g, '-').toLowerCase() || 'custom'}.pdf`)
}

function drawCropMarks(ctx: CanvasRenderingContext2D, faces: FaceSpec[]) {
  ctx.strokeStyle = '#999999'
  ctx.lineWidth = 1
  const arm = 16
  faces.forEach(f => {
    const corners = [
      [f.x, f.y], [f.x + f.w, f.y],
      [f.x, f.y + f.h], [f.x + f.w, f.y + f.h],
    ] as [number, number][]
    corners.forEach(([cx, cy]) => {
      ctx.beginPath(); ctx.moveTo(cx - arm, cy); ctx.lineTo(cx + arm, cy); ctx.stroke()
      ctx.beginPath(); ctx.moveTo(cx, cy - arm); ctx.lineTo(cx, cy + arm); ctx.stroke()
    })
  })
}
```

```tsx
// src/components/designer/BoxCanvas.tsx
'use client'
import { useEffect, useRef, RefObject } from 'react'
import type { DesignState } from '@/lib/types'
import { drawFrontFace, drawTopFace, drawSideFace, buildFrontFaceLayout, buildTopFaceLayout, buildSideFaceLayout } from '@/lib/canvas/boxRenderer'

interface Props {
  state: DesignState
  frontRef: RefObject<HTMLCanvasElement | null>
  topRef:   RefObject<HTMLCanvasElement | null>
  sideRef:  RefObject<HTMLCanvasElement | null>
}

export default function BoxCanvas({ state, frontRef, topRef, sideRef }: Props) {
  const carImageRef = useRef<HTMLImageElement | null>(null)

  // Load car image when URL changes
  useEffect(() => {
    if (!state.car_image_url) { carImageRef.current = null; redrawAll(); return }
    const img = new Image()
    img.onload = () => { carImageRef.current = img; redrawAll() }
    img.src = state.car_image_url
  }, [state.car_image_url])

  // Redraw all faces when any state changes
  useEffect(() => { redrawAll() }, [state])

  function redrawAll() {
    const front = frontRef.current
    const top   = topRef.current
    const side  = sideRef.current
    if (!front || !top || !side) return

    const fCtx = front.getContext('2d')
    const tCtx = top.getContext('2d')
    const sCtx = side.getContext('2d')
    if (!fCtx || !tCtx || !sCtx) return

    drawFrontFace(fCtx, state, carImageRef.current)
    drawTopFace(tCtx, state)
    drawSideFace(sCtx, state)
  }

  const frontDims = buildFrontFaceLayout(state.box_size)
  const topDims   = buildTopFaceLayout(state.box_size)
  const sideDims  = buildSideFaceLayout(state.box_size)

  return (
```

Canvas refs are used as `background-image` via `toDataURL()` which updates on each render.

```tsx
// src/components/designer/BoxPreview.tsx
'use client'
import { RefObject, useEffect, useState } from 'react'
import type { DesignState } from '@/lib/types'

interface Props {
  state:     DesignState
  frontRef:  RefObject<HTMLCanvasElement | null>
  topRef:    RefObject<HTMLCanvasElement | null>
  sideRef:   RefObject<HTMLCanvasElement | null>
  className?: string
}

// Box visual dimensions (px) — these are purely CSS, not canvas pixels
const DIMS = {
  minigt:  { L: 280, H: 96,  W: 56,  topH: 24 },
  poprace: { L: 260, H: 110, W: 62,  topH: 28 },
}

export default function BoxPreview({ state, frontRef, topRef, sideRef, className = '' }: Props) {
  const [frontUrl, setFrontUrl] = useState<string>('')
  const [topUrl,   setTopUrl]   = useState<string>('')
  const [sideUrl,  setSideUrl]  = useState<string>('')

  useEffect(() => {
    const update = () => {
      if (frontRef.current) setFrontUrl(frontRef.current.toDataURL())
      if (topRef.current)   setTopUrl(topRef.current.toDataURL())
      if (sideRef.current)  setSideUrl(sideRef.current.toDataURL())
    }
    // Small delay to let BoxCanvas finish drawing
    const t = setTimeout(update, 50)
    return () => clearTimeout(t)
  }, [state])

  const d = DIMS[state.box_size]
  const sceneW = d.L + d.W + 12
  const sceneH = d.H + d.topH + 12

  return (
    <div className={`flex flex-col items-center gap-3 ${className}`}>
      {/* Box scene */}
      <div
        style={{ width: sceneW, height: sceneH }}
        className="relative"
        style={{
          width: sceneW,
          height: sceneH,
          filter: 'drop-shadow(0 20px 40px rgba(0,0,0,0.22)) drop-shadow(0 4px 8px rgba(0,0,0,0.12))',
        }}
      >
        {/* Corner fill */}
        <div style={{
          position: 'absolute',
          bottom: d.H, left: d.L,
          width: d.W, height: d.topH,
          background: darkenHex(state.bg_color, 0.35),
          transformOrigin: 'bottom left',
          transform: 'skewX(-30deg) skewY(-30deg)',
        }} />

        {/* Top face */}
        <div style={{
          position: 'absolute',
          bottom: d.H, left: 0,
          width: d.L, height: d.topH,
          transformOrigin: 'bottom left',
          transform: 'skewX(-30deg)',
          backgroundImage: topUrl ? `url(${topUrl})` : undefined,
          backgroundSize: 'cover',
          backgroundColor: darkenHex(state.bg_color, 0.15),
          overflow: 'hidden',
        }} />

        {/* Right side face */}
        <div style={{
          position: 'absolute',
          bottom: 0, left: d.L,
          width: d.W, height: d.H,
          transformOrigin: 'top left',
          transform: 'skewY(-30deg)',
          backgroundImage: sideUrl ? `url(${sideUrl})` : undefined,
          backgroundSize: 'cover',
          backgroundColor: darkenHex(state.bg_color, 0.28),
          overflow: 'hidden',
        }} />

        {/* Front face */}
        <div style={{
          position: 'absolute',
          bottom: 0, left: 0,
          width: d.L, height: d.H,
          backgroundImage: frontUrl ? `url(${frontUrl})` : undefined,
          backgroundSize: 'cover',
          backgroundColor: state.bg_color,
          overflow: 'hidden',
        }} />
      </div>

      {/* Ground shadow */}
      <div style={{
        width: sceneW * 0.9,
        height: 12,
        background: 'radial-gradient(ellipse, rgba(0,0,0,0.16) 0%, transparent 70%)',
        borderRadius: '50%',
      }} />
    </div>
  )
}

function darkenHex(hex: string, amt: number): string {
  try {
    const n = parseInt(hex.replace('#', ''), 16)
    const r = Math.max(0, ((n >> 16) & 255) - Math.round(amt * 255))
    const g = Math.max(0, ((n >> 8) & 255) - Math.round(amt * 255))
    const b = Math.max(0, (n & 255) - Math.round(amt * 255))
    return `rgb(${r},${g},${b})`
  } catch { return hex }
}
```
